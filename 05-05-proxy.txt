from fastapi import FastAPI, HTTPException, Request
import httpx

app = FastAPI()

# A cheat‑sheet mapping short names to container URLs
SERVICE_MAP = {
    "assistant": "http://assistant:8000",
    "autotask": "http://autotask:8000",
    # … Add all 21 endpoints
}

@app.api_route("/{service}/{rest:path}", methods=["GET", "POST", "PUT", "DELETE"])
async def proxy(service: str, rest: str, request: Request):
    if service not in SERVICE_MAP:
        raise HTTPException(status_code=404, detail="Unknown service")
    base = SERVICE_MAP[service]
    url = f"{base}/{rest}"

    async with httpx.AsyncClient() as client:
        resp = await client.request(
            request.method,
            url,
            headers=request.headers.raw,
            content=await request.body(),
            params=request.query_params,
        )
    return resp.json()
